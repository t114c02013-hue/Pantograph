        bool CanPass(RuntimeStep s, SubAction[] subs, int idx, out bool skippedOptional)
        {
            skippedOptional = false;

            if (!subs[idx].optional) return false;

        /// <summary>
        /// 依序推進小動作：目前這個的條件成立就鎖定完成、換下一個（可連續推進）。回傳是否全部完成。
        /// 所有模式都要求（跳過小動作不能完成步驟）；教學模式才把小動作顯示出來（由 HintService／介面決定）。
        /// </summary>
        bool AdvanceSubActions(RuntimeStep s)
        {
            var subs = SubActions(s);
            int idx = (int)state.GetFloat("procedure.sub_index");
            int start = idx;
            holdReason = null;
            while (idx < subs.Length && CanPass(s, subs, idx, out bool skippedOpt))
            {
                int next = idx + 1;
                if (skippedOpt)
                {
                    // 略過：中間的選做都記 skipped，讓它略過的那個必做也已經完成，一起推過去
                    int j = NextRequired(subs, idx);
                    for (int k = idx; k < j; k++) state.Set(OptionalKey(s.RunId, k), "skipped");
                    next = Math.Min(j + 1, subs.Length);
                }
                else if (subs[idx].optional) state.Set(OptionalKey(s.RunId, idx), "done");
                state.SetBool("procedure.sub_last_skipped", skippedOpt);   // 介面：剛過去的是略過的選做（顯示「略過」，不打勾）
                while (idx < next)
                {
                    idx++;
                    if (subSnaps.Count <= idx) subSnaps.Add(state.Snapshot());   // 第 idx 個小動作從現在開始
                }
                ArmSub(s, idx);
            }
            if (idx != start) state.SetFloat("procedure.sub_index", idx);
            if (holdReason != shownHold) { state.Set("procedure.gate_reason", holdReason ?? ""); shownHold = holdReason; }   // 擋下的提示（例：扭力不足，請鎖到扳手響）
            PublishBackState();
            return idx >= subs.Length;
        }

        static bool ApplyInBatch(ISimState s, IEventBus bus, string boltId, string text, float value, string correctSocket)
        {
            var parts = (text ?? "").Split(':');
            string st = s.Get(boltId + ".state", Tight);
            switch (parts[0])
            {
                case "rotate":
                {
                    string size = parts.Length > 1 ? parts[1] : "";
                    string dir = parts.Length > 2 ? parts[2] : "loosen";
                    if (size != correctSocket) { s.SetBool(boltId + ".wrong_socket", true); return false; }
                    float deg = System.Math.Abs(value);
                    bool tight = st == Tight || (st == Torqued && !IsStripBolt(boltId));   // R1.5b b4：集電弓底座、高壓端子螺栓開場是已上扭力，拆的時候照鎖緊的鬆
                    if (dir == "loosen" && (tight || st == Loosened || st == Out))
                    {
                        float a = s.GetFloat(boltId + ".angle_deg") + deg;
                        s.SetFloat(boltId + ".angle_deg", a);
                        if (a >= OutAt) s.Set(boltId + ".state", Out);
                        else if (a >= LoosenedAt && tight) { s.Set(boltId + ".state", Loosened); AppendOrder(s, boltId, "loosen_order"); }
                        return true;
                    }
                    if (dir == "lock" && (st == Placed || st == Snug))
                    {
                        if (!IsStripBolt(boltId) && !OthersAtLeast(s, boltId, Placed) && !s.GetBool(boltId + ".tighten_before_all")) s.SetBool(boltId + ".tighten_before_all", true);   // R1.5b b4（IN_06）：四支還沒都放回就開始鎖
                        float a = s.GetFloat(boltId + ".lock_deg") + deg;
                        s.SetFloat(boltId + ".lock_deg", a);
                        if (a >= SnugAt && st == Placed) { s.Set(boltId + ".state", Snug); AppendOrder(s, boltId, "pretighten_order"); }
                        if (a > OvertightAt && !OthersAtLeast(s, boltId, Snug)) s.SetBool(boltId + ".overtight_first", true);
                        return true;
                    }
                    return false;
                }
                case "torque":
                {
                    if (st != Snug && st != Torqued) return false;
                    float.TryParse(parts.Length > 1 ? parts[1] : "0", NumberStyles.Float, CultureInfo.InvariantCulture, out var setting);
                    s.SetFloat(boltId + ".torque_setting", setting);
                    s.SetFloat(boltId + ".torque_Nm", value);
                    if (st == Snug) { s.Set(boltId + ".state", Torqued); AppendOrder(s, boltId, "torque_order"); }
                    return true;
                }
                case "mark":
                    if (st != Torqued || s.GetBool(boltId + ".marked")) return false;
                    s.SetBool(boltId + ".marked", true);
                    AppendOrder(s, boltId, "mark_order");
                    return true;
                case "take":
                    if (st != Out) return false;
                    s.Set(boltId + ".state", InTray);
                    bus?.Publish(new ActionPerformed { targetId = TrayFor(s, boltId), text = "add:bolt", pointId = boltId, time = SimClock.Now });   // 放進該側的螺栓盤
                    return true;
                case "place":
                    if (st != InTray || s.GetFloat(IsStripBolt(boltId) ? "tray.bolts" : BoltPouchCount) < 1 || !SimIds.CanPlaceBolt(s, boltId)) return false;
                    s.Set(boltId + ".state", Placed);
                    s.SetFloat(boltId + ".lock_deg", 0);
                    bus?.Publish(new ActionPerformed { targetId = TrayFor(s, boltId), text = "remove:bolt", pointId = boltId, time = SimClock.Now });   // 從該側的螺栓盤拿
                    return true;
            }
            return false;
        }

    /// <summary>terry 2026-11-15（協調者第 153 題 A）：重建場景時把三角面數寫進建置報告。含隱藏物件（日常／定期課程開關的物件也算）。</summary>
    public static class SceneTriangleReport
    {
        public const string LrvName = "LRV", RoofBoxPrefix = "Roof_", PlatformName = "ENV_InspectionPlatform";

        public static int Tris(Transform root) => root == null ? 0
            : root.GetComponentsInChildren<MeshFilter>(true).Sum(f => f.sharedMesh != null ? f.sharedMesh.triangles.Length / 3 : 0)
            + root.GetComponentsInChildren<SkinnedMeshRenderer>(true).Sum(s => s.sharedMesh != null ? s.sharedMesh.triangles.Length / 3 : 0);

        static Transform Find(IEnumerable<GameObject> roots, string name) =>
            roots.SelectMany(r => r.GetComponentsInChildren<Transform>(true)).FirstOrDefault(t => t.name == name);

        public static string Build(IEnumerable<GameObject> roots)
        {
            var list = roots.ToList();
            var lrv = Find(list, LrvName);
            int boxes = lrv == null ? 0 : lrv.GetComponentsInChildren<Transform>(true).Where(t => t.name.StartsWith(RoofBoxPrefix)).Sum(Tris);
            int car = Tris(lrv) - boxes, plat = Tris(Find(list, PlatformName)), total = list.Sum(r => Tris(r.transform));
            return $"三角面數：輕軌車體 {car}、車頂箱體 {boxes}、平台設備（{PlatformName}）{plat}、全場合計 {total}（含隱藏物件）";
        }
