## 1. EventBus 與 SimState（模組之間唯一的溝通管道）

### 1-1 EventBus：`Core/Services/EventBus.cs` 第 14–35 行

- 同步的發佈／訂閱，以事件型別（struct）當鍵。`Publish` 在呼叫者的執行緒、同一幀裡依序呼叫所有訂閱者。
- 發佈前先複製訂閱清單（`ToArray()`），所以訂閱者在處理中再訂閱、退訂或再發佈都安全。
- 重播模式下，發出的事件自動把 `isReplay` 設成 true（用反射，只在重播時付出代價），計分和紀錄看到這個旗標就略過。

```csharp
public class EventBus : IEventBus
{
    readonly Dictionary<Type, List<Delegate>> handlers = new Dictionary<Type, List<Delegate>>();

    public void Subscribe<T>(Action<T> handler) where T : struct
    {
        if (!handlers.TryGetValue(typeof(T), out var list)) handlers[typeof(T)] = list = new List<Delegate>();
        if (!list.Contains(handler)) list.Add(handler);
    }

    public void Unsubscribe<T>(Action<T> handler) where T : struct
    {
        if (handlers.TryGetValue(typeof(T), out var list)) list.Remove(handler);
    }

    public void Publish<T>(T evt) where T : struct
    {
        if (!handlers.TryGetValue(typeof(T), out var list) || list.Count == 0) return;
        if (SimSession.IsReplay) evt = ReplayMark<T>.Apply(evt);
        foreach (var h in list.ToArray()) ((Action<T>)h)(evt);
    }
    // … ReplayMark<T>：反射把 isReplay 欄位設成 true
}
```

### 1-2 SimState 讀寫：`Core/Services/SimState.cs` 第 41–111 行（節錄）

- 全部狀態是 `Dictionary<string,string>`（數值用 InvariantCulture 字串化）。每個鍵另記 `Version`（寫入次數，值沒變也加 1）。
- 不在批次中：`Set` 立刻生效並發一筆 `StateChanged`；值沒變則發 `StateTouched`。
- 在批次中：寫進 `pending`，讀取時「讀得到自己剛寫的」（read-your-writes），等批次結束才一起生效。

```csharp
public class SimState : ISimState
{
    readonly Dictionary<string, string> values = new Dictionary<string, string>();
    readonly Dictionary<string, long> versions = new Dictionary<string, long>();
    readonly Dictionary<string, string> pending = new Dictionary<string, string>();   // 批次暫存（null＝刪除）
    readonly List<string> pendingOrder = new List<string>();
    int depth;   // 批次巢狀深度
    // …

    bool TryRead(string key, out string value)
    {
        if (depth > 0 && pending.TryGetValue(key, out value)) return value != null;   // read-your-writes
        return values.TryGetValue(key, out value);
    }

    public string Get(string key, string fallback = "") => TryRead(key, out var v) ? v : fallback;
    public float GetFloat(string key, float fallback = 0f) =>
        TryRead(key, out var v) && float.TryParse(v, NumberStyles.Float, CultureInfo.InvariantCulture, out var f) ? f : fallback;
    public bool GetBool(string key) => TryRead(key, out var v) && v == "true";

    public void Set(string key, string value, bool force)
    {
        if (depth > 0)
        {
            if (!pending.ContainsKey(key)) pendingOrder.Add(key);
            pending[key] = value;
            if (force) forced.Add(key);
            return;
        }
        // 不在批次：等於只有一筆的批次
        var change = Apply(key, value, force);
        if (change.HasValue) Publish(new[] { change.Value }, false);
        else PublishTouched(new[] { key });
    }
}
```

### 1-3 SimState 批次與 StateChanged：同檔第 140–215 行（節錄）

- `BeginBatch()` 回傳 `IDisposable`，`using` 結束時呼叫 `End()`。支援巢狀，只有最外層結束時才 Commit。
- Commit 時依第一次寫入的順序套用，所有變更合成**一個** `StateChanged`（帶 `Changes` 清單）。訂閱者收到時，整批都已生效，不會看到寫到一半的狀態。
- `Batch(Action)` 版本：例外時整批放棄，一個鍵都不寫。`AfterCommit` 讓「依賴這批結果的事件」在 Commit 後才發。

```csharp
public IDisposable BeginBatch() { depth++; return new Scope(this); }

public void Batch(Action body)
{
    using (BeginBatch())
    {
        try { body(); }
        catch (Exception ex) { aborted = true; ErrorLog("[SimState] 批次寫入中發生例外，整批放棄…" + ex); throw; }
    }
}

void End()
{
    if (depth == 0) return;
    if (--depth > 0) return;                       // 巢狀：只有最外層 Commit
    var after = afterCommit.Count > 0 ? afterCommit.ToArray() : null;
    afterCommit.Clear();
    bool discard = aborted; aborted = false;
    if (pendingOrder.Count > 0)
    {
        var order = pendingOrder.ToArray();
        var batch = new Dictionary<string, string>(pending);
        var force = new HashSet<string>(forced);
        pending.Clear(); pendingOrder.Clear(); forced.Clear();
        if (discard) return;
        var changes = new List<StateChange>(); var touched = new List<string>();
        foreach (var key in order)
        {
            var c = Apply(key, batch[key], force.Contains(key));
            if (c.HasValue) changes.Add(c.Value); else touched.Add(key);
        }
        if (changes.Count > 0) Publish(changes, false);          // 一批只發一個 StateChanged
        if (touched.Count > 0) PublishTouched(touched.ToArray());
    }
    // …
    foreach (var a in after) a();                                // AfterCommit 的動作
}

StateChange? Apply(string key, string value, bool force)
{
    versions[key] = Version(key) + 1;
    bool had = values.TryGetValue(key, out var old);
    if (had && old == value || !had && value == null) return force && had ? new StateChange(key, old, old, true) : (StateChange?)null;
    if (value == null) values.Remove(key); else values[key] = value;
    return new StateChange(key, had ? old : null, value, force);
}
```

```csharp
public class CompositeActionGuard : IActionGuard
```

```csharp
void Expand()
{
    var rb = procedure.repeatBlock;
    var defs = procedure.steps;
    int from = rb != null && !string.IsNullOrEmpty(rb.fromStepId) ? defs.FindIndex(s => s.stepId == rb.fromStepId) : -1;
    int to = rb != null && !string.IsNullOrEmpty(rb.toStepId) ? defs.FindIndex(s => s.stepId == rb.toStepId) : -1;
    for (int i = 0; i < defs.Count; i++)
    {
        if (from >= 0 && to >= from && i == from)
        {
            for (int s = 0; s < rb.sides.Length; s++)          // 先整段 L，再整段 R
                for (int j = from; j <= to; j++)
                    steps.Add(new RuntimeStep { def = defs[j], defIndex = j + 1, repeated = true, side = rb.sides[s], sideLabel = /* … */ });
            i = to;
            continue;
        }
        string fixedSide = defs[i].side == "L" || defs[i].side == "R" ? defs[i].side : "";
        steps.Add(new RuntimeStep { def = defs[i], defIndex = i + 1, side = fixedSide, /* … */ });
    }
}
```

### 3-3 小動作判定：第 315–331 行

- 每一步有數個「小動作」（介面上的打勾項），**照順序**判定：從 `procedure.sub_index` 開始，條件成立就往下，遇到第一個不成立就停。
- 已打勾的不會因為條件之後變回去而取消。很多小動作的條件本來就會在同一步裡變回去，例如「拿起扳手」在放回後就不成立，但那個勾不該取消。
- 每推進一個小動作就存一份快照（`subSnaps`），給「↶ 上一步」用。

```csharp
bool AdvanceSubActions(RuntimeStep s)
{
    var subs = SubActions(s);
    int idx = (int)state.GetFloat("procedure.sub_index");
    int start = idx;
    holdReason = null;
    while (idx < subs.Length && SubDone(s, subs[idx]))
    {
        idx++;
        ArmSub(s, idx);
        if (subSnaps.Count <= idx) subSnaps.Add(state.Snapshot());   // 第 idx 個小動作從現在開始
    }
    if (idx != start) state.SetFloat("procedure.sub_index", idx);
    if (holdReason != shownHold) { state.Set("procedure.gate_reason", holdReason ?? ""); shownHold = holdReason; }
    PublishBackState();
    return idx >= subs.Length;
}
```

### 3-4 步驟完成判定：第 372–411 行

- 完成＝小動作全勾**且**步驟的完成條件（`completion`，見第 4 節）成立**且**所有步驟閘門（例 SafetyGuard）同意。
- 一次判定可能連續完成好幾步（內層 `do…while`）。判定中又收到 `StateChanged` 時，不重入，記 `evaluateAgain`，本輪結束後再判一次；超過 `MaxRounds`（8）視為狀態不穩定，記 Error。
- 狀態穩定後才存「本步開始快照」，給「重做本步」用。

```csharp
public void Evaluate()
{
    if (!Started || Finished) return;
    if (state.InBatch) return;                       // 批次還沒 Commit：等 Commit 的 StateChanged 再判定
    if (evaluating) { evaluateAgain = true; return; }
    evaluating = true;
    int rounds = 0;
    try
    {
        do
        {
            evaluateAgain = false;
            do
            {
                var s = Current;
                if (s == null) break;
                bool subsDone = AdvanceSubActions(s);
                if (!subsDone || !RuleEvaluators.Evaluate(s.def.completion, ContextFor(s))) break;
                if (gate != null && !gate.CanComplete(s.RunId, out var reason)) { /* 寫 procedure.gate_reason */ break; }
                Complete(s);                         // 寫 step.X.done、發 StepCompleted、MoveTo 下一步
            } while (!Finished);
            if (evaluateAgain && ++rounds >= MaxRounds) { ErrorLog($"… 重新判定超過 {MaxRounds} 次狀態仍不穩定"); break; }
        } while (evaluateAgain && !Finished && !state.InBatch);
        if (snapshotPending && !evaluateAgain && !state.InBatch && Current != null)
        {
            snapshotPending = false;
            snapshot = state.Snapshot();             // 本步開始快照（重做用）
            // …
        }
    }
    finally { evaluating = false; evaluateAgain = false; }
}
```

### 3-6 卡關偵測（防護網，只記錄不推進）：第 427–475 行

- App 每幀呼叫 `Poll`，內部每 0.5 秒檢查一次。
- `missed_commit`：條件都成立但步驟沒推進，代表判定漏觸發，是 bug，記 Error。
- `rule_mismatch`：小動作全勾，但完成條件超過 3 秒仍不成立。把「還差什麼」寫到 `procedure.stall_*`，介面據此提示，「帶我過去」也帶到那裡。每步每種只記一次。

```csharp
public void Poll(float now)
{
    if (!Started || Finished || evaluating || state.InBatch) return;
    if (now < nextPoll) return;
    nextPoll = now + PollInterval;                                    // 0.5 s
    var s = Current; if (s == null) return;
    var subs = SubActions(s);
    int idx = (int)state.GetFloat("procedure.sub_index");
    int would = idx;
    while (would < subs.Length && SubDone(s, subs[would])) { would++; if (FreshMode) break; }
    var ctx = ContextFor(s);
    bool rule = RuleEvaluators.Evaluate(s.def.completion, ctx);
    string gateReason = "";
    bool gateOk = gate == null || gate.CanComplete(s.RunId, out gateReason);

    if (would > idx || (would >= subs.Length && rule && gateOk))
    {
        Report(s, "missed_commit", RuleExplainer.Explain(s.def.completion, ctx), now, /* … */);
        return;
    }
    if (idx < subs.Length) { allDoneSince = -1f; return; }
    if (allDoneSince < 0f) { allDoneSince = now; return; }
    if (now - allDoneSince < StallSeconds) return;                   // 3 s
    var unmet = RuleExplainer.Explain(s.def.completion, ctx);
    if (!gateOk) unmet.Insert(0, new UnmetCondition { id = "Gate:" + gateReason });
    UpdateStallHint(s, subs, unmet);                                  // procedure.stall_text／stall_target／…
    Report(s, "rule_mismatch", unmet, now, "");
}
```

```csharp
public class EvalContext
{
    public ISimState state;
    public string side;
    public Func<bool> allPreviousDone = () => true;
    public string K(string pattern, int n = 0) => StateKeys.Expand(pattern, side, n);
}

public class StateEqualsEvaluator : Evaluator<StateEquals>
{
    protected override bool Eval(StateEquals r, EvalContext c) => c.state.Get(c.K(r.key)) == r.value;
}

public class ChecklistAllEvaluator : Evaluator<ChecklistAll>
{
    protected override bool Eval(ChecklistAll r, EvalContext c) => r.keys.All(k => c.state.GetBool(c.K(k)));
}

public class AllOfEvaluator : Evaluator<AllOf>
{
    protected override bool Eval(AllOf r, EvalContext c) => r.rules.All(x => RuleEvaluators.Evaluate(x, c));
}

public static class RuleEvaluators
{
    static readonly Dictionary<Type, ICompletionEvaluator> map = new ICompletionEvaluator[]
    {
        new StateEqualsEvaluator(), new ChecklistAllEvaluator(), new OrderedSequenceEvaluator(), new ValueInRangeEvaluator(),
        new DecisionEvaluator(), new ReadingMatchesEvaluator(), new ProcedureDoneEvaluator(), new AllOfEvaluator()
    }.ToDictionary(e => e.RuleType);

    public static bool Evaluate(CompletionRule rule, EvalContext ctx)
    {
        if (rule == null) return false;
        if (!map.TryGetValue(rule.GetType(), out var ev)) throw new InvalidOperationException("沒有對應的判定器：" + rule.GetType().Name);
        return ev.Evaluate(rule, ctx);
    }
}
```

## 6. 螺栓狀態機（`Modules/Parts/_Common/FastenerLogic.cs`）

- 純靜態函式，零件元件和測試共用。狀態：`Tight → Loosened → Out → InTray → Placed → Snug → Torqued`（另有 `marked`）。
- 整個 `Apply` 包在一個批次裡：state、角度、扭力、順序鍵同時生效，不會出現「扭力到了、狀態還沒變」的中間態。
- 數值（鬆開 150°、脫出 720°、預鎖 360°、過鎖 540°）是示例值，常數在第 16 行。

第 15–58、84–98 行（節錄）：

```csharp
public const string Tight = "Tight", Loosened = "Loosened", Out = "Out", InTray = "InTray", Placed = "Placed", Snug = "Snug", Torqued = "Torqued";
public const float LoosenedAt = 150f, OutAt = 720f, SnugAt = 360f, OvertightAt = 540f;

public static bool Apply(ISimState s, IEventBus bus, string boltId, string text, float value, string correctSocket) =>
    s.BatchResult(() => ApplyInBatch(s, bus, boltId, text, value, correctSocket));

static bool ApplyInBatch(ISimState s, IEventBus bus, string boltId, string text, float value, string correctSocket)
{
    var parts = (text ?? "").Split(':');                  // 例 "rotate:13:loosen"、"torque:25"、"take"、"place"、"mark"
    string st = s.Get(boltId + ".state", Tight);
    switch (parts[0])
    {
        case "rotate":
        {
            string size = parts.Length > 1 ? parts[1] : "", dir = parts.Length > 2 ? parts[2] : "loosen";
            if (size != correctSocket) { s.SetBool(boltId + ".wrong_socket", true); return false; }
            float deg = Math.Abs(value);
            bool tight = st == Tight || (st == Torqued && !IsStripBolt(boltId));
            if (dir == "loosen" && (tight || st == Loosened || st == Out))
            {
                float a = s.GetFloat(boltId + ".angle_deg") + deg;
                s.SetFloat(boltId + ".angle_deg", a);
                if (a >= OutAt) s.Set(boltId + ".state", Out);
                else if (a >= LoosenedAt && tight) { s.Set(boltId + ".state", Loosened); AppendOrder(s, boltId, "loosen_order"); }
                return true;
            }
            if (dir == "lock" && (st == Placed || st == Snug))
            {
                float a = s.GetFloat(boltId + ".lock_deg") + deg;
                s.SetFloat(boltId + ".lock_deg", a);
                if (a >= SnugAt && st == Placed) { s.Set(boltId + ".state", Snug); AppendOrder(s, boltId, "pretighten_order"); }
                if (a > OvertightAt && !OthersAtLeast(s, boltId, Snug)) s.SetBool(boltId + ".overtight_first", true);
                return true;
            }
            return false;
        }
        case "torque": /* Snug → Torqued，記 torque_Nm、torque_setting、torque_order */
        case "mark":   /* Torqued 才能畫防鬆標記 */
        case "take":
            if (st != Out) return false;
            s.Set(boltId + ".state", InTray);
            bus?.Publish(new ActionPerformed { targetId = TrayFor(s, boltId), text = "add:bolt", pointId = boltId, time = SimClock.Now });
            return true;
        case "place":
            // 第 13 步修正：滑板螺栓要等新滑板裝上（NewMounted）才能裝回孔位
            if (st != InTray || s.GetFloat(IsStripBolt(boltId) ? "tray.bolts" : BoltPouchCount) < 1 || !SimIds.CanPlaceBolt(s, boltId)) return false;
            s.Set(boltId + ".state", Placed);
            s.SetFloat(boltId + ".lock_deg", 0);
            bus?.Publish(new ActionPerformed { targetId = TrayFor(s, boltId), text = "remove:bolt", pointId = boltId, time = SimClock.Now });
            return true;
    }
    return false;
}
```

### 9-1 背景寫檔：`Scripts/RecordingWriter.cs` 第 13–73 行（節錄）

- 主執行緒只把「寫檔工作」放進 `ConcurrentQueue`，背景執行緒依序執行，VR 不會因為硬碟 I/O 掉幀。每行寫完就 flush，當機也保留到最後一行。

```csharp
public class RecordingWriter : IDisposable
{
    readonly ConcurrentQueue<Action> queue = new ConcurrentQueue<Action>();
    readonly AutoResetEvent signal = new AutoResetEvent(false);
    readonly Thread thread;
    volatile bool stopping;
    StreamWriter session;          // session.jsonl
    FileStream posesStream, snapStream;
    // …
    void Enqueue(Action a)
    {
        if (stopping) return;
        Interlocked.Increment(ref pending);
        idle.Reset();
        queue.Enqueue(a);
        signal.Set();
    }

    void Loop()
    {
        while (true)
        {
            while (queue.TryDequeue(out var a))
            {
                try { a(); } catch (Exception e) { LastError = e.Message; }
                if (Interlocked.Decrement(ref pending) == 0) idle.Set();
            }
            if (stopping && queue.IsEmpty) break;
            signal.WaitOne(200);
        }
    }

    public void WriteLine(string json) => Enqueue(() => { session.Write(json); session.Write('\n'); session.Flush(); });
}
```
